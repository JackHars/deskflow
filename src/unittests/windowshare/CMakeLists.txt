cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(deskflow_windowshare_core_tests LANGUAGES CXX)

enable_testing()
option(WINDOWSHARE_SANITIZERS "Enable ASan/UBSan" OFF)
add_executable(windowshare-core-tests CoreTests.cpp ../../lib/windowshare/Core.cpp)
target_include_directories(windowshare-core-tests PRIVATE ../../lib/windowshare)
target_compile_features(windowshare-core-tests PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(windowshare-core-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(WINDOWSHARE_SANITIZERS)
    target_compile_options(windowshare-core-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(windowshare-core-tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_test(NAME windowshare-core COMMAND windowshare-core-tests)

find_package(Threads REQUIRED)
foreach(check original-input viewport-lease module-policy library-constraint)
  if(check STREQUAL "original-input")
    set(foundation_sources ../../lib/windowshare/OriginalInputTests.cpp
      ../../lib/windowshare/OriginalInput.cpp ../../lib/windowshare/OriginalKeyboard.cpp)
  elseif(check STREQUAL "viewport-lease")
    set(foundation_sources ../../lib/windowshare/ViewportLeaseTests.cpp ../../lib/windowshare/ViewportLease.cpp)
  elseif(check STREQUAL "library-constraint")
    set(foundation_sources ../../lib/windowshare/LocalInputLibraryConstraintTests.cpp
      ../../lib/windowshare/LocalInputLibraryConstraint.cpp)
  else()
    set(foundation_sources ../../lib/windowshare/LocalInputModulePolicyTests.cpp
      ../../lib/windowshare/LocalInputModulePolicy.cpp)
  endif()
  add_executable(windowshare-${check}-tests ${foundation_sources})
  target_include_directories(windowshare-${check}-tests PRIVATE ../../lib/windowshare)
  target_compile_features(windowshare-${check}-tests PRIVATE cxx_std_20)
  target_link_libraries(windowshare-${check}-tests PRIVATE Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(windowshare-${check}-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(WINDOWSHARE_SANITIZERS)
      target_compile_options(windowshare-${check}-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(windowshare-${check}-tests PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME windowshare-${check} COMMAND windowshare-${check}-tests)
  set_tests_properties(windowshare-${check} PROPERTIES TIMEOUT 30)
endforeach()
