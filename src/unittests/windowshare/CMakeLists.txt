cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(deskflow_windowshare_core_tests LANGUAGES CXX)
endif()
enable_testing()
option(WINDOWSHARE_SANITIZERS "Enable ASan/UBSan" OFF)
add_executable(windowshare-core-tests CoreTests.cpp ../../lib/windowshare/Core.cpp)
target_include_directories(windowshare-core-tests PRIVATE ../../lib/windowshare)
target_compile_features(windowshare-core-tests PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(windowshare-core-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(WINDOWSHARE_SANITIZERS)
    target_compile_options(windowshare-core-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(windowshare-core-tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_test(NAME windowshare-core COMMAND windowshare-core-tests)

add_executable(windowshare-platform-markers-tests PlatformMarkersTests.cpp)
target_include_directories(windowshare-platform-markers-tests PRIVATE ../../lib/windowshare)
target_compile_features(windowshare-platform-markers-tests PRIVATE cxx_std_17)
if(APPLE)
  target_link_libraries(windowshare-platform-markers-tests PRIVATE "-framework ApplicationServices")
endif()
add_test(NAME windowshare-platform-markers COMMAND windowshare-platform-markers-tests)

add_executable(windowshare-input-routes-tests InputRouteStateTests.cpp)
target_include_directories(windowshare-input-routes-tests PRIVATE ../../lib/windowshare)
target_compile_features(windowshare-input-routes-tests PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(windowshare-input-routes-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(WINDOWSHARE_SANITIZERS)
    target_compile_options(windowshare-input-routes-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(windowshare-input-routes-tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_test(NAME windowshare-input-routes COMMAND windowshare-input-routes-tests)
