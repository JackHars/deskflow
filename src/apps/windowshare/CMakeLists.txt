cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(deskflow_windowshare LANGUAGES CXX)
  enable_testing()
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
set(WS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../lib/windowshare")
add_library(windowshare-selection STATIC
  "${WS_ROOT}/SelectedPeerMailbox.cpp" "${WS_ROOT}/SelectedPeerPublisher.cpp")
target_compile_features(windowshare-selection PUBLIC cxx_std_20)
target_include_directories(windowshare-selection PUBLIC "${WS_ROOT}" "${WS_ROOT}/..")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(windowshare-selection PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(windowshare-transport STATIC Transport.cpp KeyboardRouteProtocol.cpp "${WS_ROOT}/Core.cpp")
target_compile_features(windowshare-transport PUBLIC cxx_std_20)
target_compile_definitions(windowshare-transport PUBLIC QT_NO_KEYWORDS)
target_include_directories(windowshare-transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${WS_ROOT}")
target_link_libraries(windowshare-transport PUBLIC Qt6::Core Qt6::Network)
if(APPLE)
  target_link_libraries(windowshare-transport PRIVATE "-framework Security" "-framework CoreFoundation")
endif()
option(WINDOWSHARE_TRANSPORT_TESTS "Build Qt transport tests" ON)
if(APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # A private inherited socketpair and explicit packaged companion bootstrap.
  # Merely linking this library neither launches a companion nor enables input.
  add_library(windowshare-local-input STATIC
    "${WS_ROOT}/LocalInputChannel.cpp" "${WS_ROOT}/OriginalKeyboard.cpp"
    "${WS_ROOT}/LocalInputBootstrap.cpp" "${WS_ROOT}/LocalInputCompanion.cpp"
    "${WS_ROOT}/LocalInputSession.cpp")
  target_compile_features(windowshare-local-input PUBLIC cxx_std_20)
  target_compile_definitions(windowshare-local-input PUBLIC QT_NO_KEYWORDS WINDOWSHARE_LOCAL_INPUT=1)
  target_include_directories(windowshare-local-input PUBLIC "${WS_ROOT}")
  target_link_libraries(windowshare-local-input PUBLIC Qt6::Core Qt6::Network windowshare-selection)
  if(APPLE)
    enable_language(OBJCXX)
    target_sources(windowshare-local-input PRIVATE "${WS_ROOT}/LocalInputPeerTrust_mac.mm")
    set_target_properties(windowshare-local-input PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
    target_link_libraries(windowshare-local-input PRIVATE "-framework Security" "-framework CoreFoundation")
  else()
    target_sources(windowshare-local-input PRIVATE "${WS_ROOT}/LocalInputPeerTrust_stub.cpp")
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(windowshare-local-input PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
if(WINDOWSHARE_TRANSPORT_TESTS)
  enable_testing()
  if(TARGET windowshare-local-input)
    add_executable(windowshare-local-input-tests "${WS_ROOT}/LocalInputChannelTests.cpp")
    target_link_libraries(windowshare-local-input-tests PRIVATE windowshare-local-input)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(windowshare-local-input-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME windowshare-local-input COMMAND windowshare-local-input-tests)
    set_tests_properties(windowshare-local-input PROPERTIES TIMEOUT 60)
    add_executable(windowshare-bootstrap-tests "${WS_ROOT}/LocalInputBootstrapTests.cpp")
    target_link_libraries(windowshare-bootstrap-tests PRIVATE windowshare-local-input)
    add_test(NAME windowshare-bootstrap COMMAND windowshare-bootstrap-tests)
    set_tests_properties(windowshare-bootstrap PROPERTIES TIMEOUT 30)
    # Substitute only the fixture verifier. Never link the production verifier
    # into this test, and never compile its substitution into either executable.
    add_executable(windowshare-companion-tests
      "${WS_ROOT}/LocalInputCompanionTests.cpp" "${WS_ROOT}/LocalInputCompanion.cpp"
      "${WS_ROOT}/LocalInputChannel.cpp" "${WS_ROOT}/OriginalKeyboard.cpp" "${WS_ROOT}/LocalInputSession.cpp")
    target_compile_features(windowshare-companion-tests PRIVATE cxx_std_20)
    target_compile_definitions(windowshare-companion-tests PRIVATE QT_NO_KEYWORDS LOCAL_INPUT_COMPANION_TEST_VERIFIER=1)
    target_link_libraries(windowshare-companion-tests PRIVATE Qt6::Core Qt6::Network windowshare-selection)
    add_test(NAME windowshare-companion COMMAND windowshare-companion-tests)
    set_tests_properties(windowshare-companion PROPERTIES TIMEOUT 60)
    if(APPLE)
      add_executable(windowshare-peer-trust-tests "${WS_ROOT}/LocalInputPeerTrustTests.cpp")
      target_link_libraries(windowshare-peer-trust-tests PRIVATE windowshare-local-input)
      add_test(NAME windowshare-peer-trust COMMAND windowshare-peer-trust-tests)
      set_tests_properties(windowshare-peer-trust PROPERTIES TIMEOUT 60)
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(windowshare-bootstrap-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
      target_compile_options(windowshare-companion-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
      if(TARGET windowshare-peer-trust-tests)
        target_compile_options(windowshare-peer-trust-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
      endif()
    endif()
  endif()
  add_executable(windowshare-transport-tests TransportTests.cpp)
  target_link_libraries(windowshare-transport-tests PRIVATE windowshare-transport)
  add_test(NAME windowshare-transport COMMAND windowshare-transport-tests)
  set_tests_properties(windowshare-transport PROPERTIES TIMEOUT 60)
  if(APPLE)
    # The native backend has different certificate error reporting from OpenSSL.
    add_test(NAME windowshare-transport-securetransport COMMAND windowshare-transport-tests --tls-backend=securetransport)
    set_tests_properties(windowshare-transport-securetransport PROPERTIES TIMEOUT 60)
  endif()
endif()

# Opt-in also for standalone builds: transport tests need no capture permissions.
option(WINDOWSHARE_APP "Build the experimental native host and viewer" OFF)
if(WINDOWSHARE_APP)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(WS_AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
  add_library(windowshare-viewer STATIC Viewer.cpp VideoDecoder.cpp)
  target_link_libraries(windowshare-viewer PUBLIC windowshare-transport Qt6::Widgets PkgConfig::WS_AV)
  add_library(windowshare-native STATIC)
  target_link_libraries(windowshare-native PUBLIC windowshare-transport Qt6::Widgets)
  if(APPLE)
    enable_language(OBJCXX)
    add_executable(deskflow-holding-display-helper HoldingDisplayHelper.cpp HoldingDisplay.mm)
    target_compile_definitions(deskflow-holding-display-helper PRIVATE DESKFLOW_HOLDING_DISPLAY_HELPER=1)
    target_compile_options(deskflow-holding-display-helper PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(deskflow-holding-display-helper PRIVATE windowshare-transport "-framework Foundation" "-framework CoreGraphics")
    set_target_properties(deskflow-holding-display-helper PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
    target_sources(windowshare-native PRIVATE NativeHost_mac.mm HoldingDisplay.mm)
    target_compile_options(windowshare-native PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    set_target_properties(windowshare-native PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
    foreach(framework AppKit ApplicationServices ScreenCaptureKit VideoToolbox CoreMedia CoreVideo Carbon)
      find_library(WS_${framework} ${framework} REQUIRED)
      target_link_libraries(windowshare-native PRIVATE "${WS_${framework}}")
    endforeach()
  else()
    target_sources(windowshare-native PRIVATE NativeHost_stub.cpp)
  endif()
  add_executable(deskflow-windowshare main.cpp)
  target_link_libraries(deskflow-windowshare PRIVATE windowshare-native windowshare-viewer)
  if(TARGET windowshare-local-input)
    target_link_libraries(deskflow-windowshare PRIVATE windowshare-local-input)
  endif()
  if(APPLE)
    set_target_properties(deskflow-windowshare PROPERTIES
      MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_GUI_IDENTIFIER org.deskflow.windowshare.experimental
      MACOSX_BUNDLE_BUNDLE_NAME "Deskflow Window Share"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
    add_dependencies(deskflow-windowshare deskflow-holding-display-helper)
    add_custom_command(TARGET deskflow-windowshare POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:deskflow-holding-display-helper>
        $<TARGET_FILE_DIR:deskflow-windowshare>/deskflow-holding-display-helper)
    # A helper-only rebuild must also refresh an already linked application.
    add_custom_target(windowshare-bundle-helper ALL
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:deskflow-holding-display-helper>
        $<TARGET_FILE_DIR:deskflow-windowshare>/deskflow-holding-display-helper
      DEPENDS deskflow-windowshare deskflow-holding-display-helper)
  endif()
  install(TARGETS deskflow-windowshare BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  if(WINDOWSHARE_TRANSPORT_TESTS)
    add_executable(windowshare-viewer-tests ViewerTests.cpp)
    target_link_libraries(windowshare-viewer-tests PRIVATE windowshare-viewer)
    add_test(NAME windowshare-viewer COMMAND windowshare-viewer-tests)
    set_tests_properties(windowshare-viewer PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(windowshare-bridge-tests BridgeTests.cpp)
    target_link_libraries(windowshare-bridge-tests PRIVATE windowshare-viewer)
    add_test(NAME windowshare-bridge COMMAND windowshare-bridge-tests)
    set_tests_properties(windowshare-bridge PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(APPLE)
      add_test(NAME windowshare-bridge-securetransport COMMAND windowshare-bridge-tests --tls-backend=securetransport)
      set_tests_properties(windowshare-bridge-securetransport PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
      add_executable(windowshare-smoke-source NativeSmokeSource.cpp)
      target_link_libraries(windowshare-smoke-source PRIVATE Qt6::Widgets)
      add_executable(windowshare-native-smoke NativeSmoke.cpp)
      target_link_libraries(windowshare-native-smoke PRIVATE windowshare-native windowshare-viewer)
      add_dependencies(windowshare-native-smoke windowshare-smoke-source)
      add_executable(windowshare-holding-tests HoldingDisplayTests.mm)
      target_link_libraries(windowshare-holding-tests PRIVATE windowshare-native)
      add_dependencies(windowshare-holding-tests deskflow-holding-display-helper)
      target_compile_options(windowshare-holding-tests PRIVATE -fobjc-arc)
      set_target_properties(windowshare-holding-tests PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
      add_test(NAME windowshare-holding COMMAND windowshare-holding-tests)
      add_executable(windowshare-native-tests NativeHostTests.cpp)
      target_link_libraries(windowshare-native-tests PRIVATE windowshare-native)
      add_test(NAME windowshare-native-recovery COMMAND windowshare-native-tests)
      add_dependencies(windowshare-native-smoke deskflow-holding-display-helper)
      # Run explicitly on the actual desktop after permission approval, not CI.
    endif()
  endif()
endif()
